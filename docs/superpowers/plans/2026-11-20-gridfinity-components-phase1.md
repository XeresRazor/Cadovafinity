> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the shared base-grid and hole modules to `GridfinityRebuilt`, create the `GridfinityCustomizers` package with the baseplate customizer, and publish several customizers under one GitHub Pages site (lidded bin at `bins/lidded/`, baseplate at `baseplate/`, a redirect at the site root).

**Architecture:** Each OpenSCAD module of `gridfinity-rebuilt-utility.scad` and `gridfinity-rebuilt-baseplate.scad` becomes a Swift function or method on `Dimensions` or a small configuration value, verified against OpenSCAD output with the existing reference method. The new package follows `LiddedBin`'s pattern: plain configuration with `validated()`, an input type, a design value, a thin executable. The workflow builds each customizer in a parallel matrix job from one layout file and assembles one Pages site.

**Tech Stack:** Swift 6.3 and Cadova 1.3, Swift Testing, OpenSCAD, Python 3 stdlib, bash, GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-11-20-gridfinity-components-design.md`. Inventory: `docs/superpowers/notes/2026-11-20-gridfinity-component-inventory.md`. Earlier plans: `2026-11-17-library-core.md`, `2026-11-18-lidded-bin.md`, `2026-11-18-customizer.md`.

```swift
import Cadova
import Foundation

/// `pattern_circular`: `n` copies of `geometry`, rotated by `i * 360 / n` for `i` in `1...n`.
public func patternCircular(count n: Int, _ geometry: any Geometry3D) -> any Geometry3D {
    Union {
        for i in 1...max(n, 1) {
            geometry.rotated(z: Angle(degrees: Double(i) * 360 / Double(max(n, 1))))
        }
    }
}

/// `copy_mirror`: the geometry plus its mirror image across the plane normal to `axis`.
public func copyMirror(_ geometry: any Geometry3D, along axis: Axis3D = .y) -> any Geometry3D {
    Union {
        geometry
        geometry.flipped(along: [axis])
    }
}

/// `rounded_square`: a square rounded rectangle.
public func roundedSquare(length: Double, height: Double, radius: Double) -> any Geometry3D {
    roundedRectangle(length: length, width: length, height: height, radius: radius)
}

/// `clp`.
public func clamp(_ x: Double, _ a: Double, _ b: Double) -> Double { min(max(x, a), b) }
```

Check Cadova's spellings for `Angle(degrees:)`, the axis type accepted by `flipped(along:)` and the 3D axis type name, and adjust. In `PatternLinear.swift` add the `Double` overload next to the `Int` one:

```swift
public func patternLinear(
    x: Double, y: Double, spacingX sx: Double, spacingY sy: Double? = nil,
    _ geometry: any Geometry3D
) -> any Geometry3D {
    let yy = (sy ?? sx) <= 0 ? sx : (sy ?? sx)
    let nx = max(Int(x.rounded(.up)), 0), ny = max(Int(y.rounded(.up)), 0)
    return Union {
        for i in 0..<nx {
            for j in 0..<ny {
                geometry.translated(x: Double(i) * sx, y: Double(j) * yy)
            }
        }
    }
    .translated(x: -(x - 1) * sx / 2, y: -(y - 1) * yy / 2)
}
```

```swift
extension Dimensions {
    /// `block_base_hole`: magnet (style 1), magnet and screw (2), magnet with a printable slit and screw (3).
    public func blockBaseHole(style: Int, offset o: Double = 0) -> any Geometry3D {
        let r1 = rHole1 - o / 2
        let r2 = rHole2 - o / 2
        let magnetHeight = 2 * (hHole - o + (style == 3 ? hSlit : 0))
        let magnet = Cylinder(radius: r2, height: magnetHeight).aligned(at: .centerZ)
            .subtracting {
                if style == 3 {
                    copyMirror(
                        Box([r2 * 3, r2 * 3, 10]).translated(x: -1.5 * r2, y: r1 + 0.1, z: hHole - o),
                        along: .y
                    )
                }
            }
        return Union {
            magnet
            if style > 1 {
                Cylinder(radius: r1, height: 2 * hBase - o).aligned(at: .centerZ)
            }
        }
    }

    /// `refined_hole`: the squeezed magnet hole with a poke-through slot (Printables @grizzie17).
    public func refinedHole() -> any Geometry3D {
        let r = rHole2 - 0.32
        let m = 2.0, mh = m - 0.1
        let ptl = hSlit * 3, pth = mh + ptl, ptr = 2.5
        return Union {
            Union {
                Box([1, r * 2, mh]).translated(x: 10, y: -r)
                Cylinder(radius: r, height: 1.9)
            }.convexHull()
            Union {
                Box([1, ptr, pth]).translated(x: -9 + 5.60, y: -ptr / 2, z: -ptl)
                Cylinder(diameter: ptr, height: pth).translated(x: -12.53 + 5.60, z: -ptl)
            }.convexHull()
        }
    }
}
```

```swift
public func gridfinityBase(
    gridX gx: Double, gridY gy: Double, length: Double? = nil,
    divisionX dx: Double = 0, divisionY dy: Double = 0,
    styleHole: Int, offset: Double = 0, finalCut: Bool = true, onlyCorners: Bool = false
) -> any Geometry3D {
    let l = length ?? lGrid
    func divisions(_ g: Double, _ d: Double) -> Double {
        let counts = (1...5).filter { i in
            let f = (abs(g * Double(i))).truncatingRemainder(dividingBy: 1)
            return f < 0.001 || f > 0.999
        }
        return 1 / (d == 0 ? Double(counts.first ?? 1) : d.rounded())
    }
    let dbnx = divisions(gx, dx), dbny = divisions(gy, dy)
    let xx = gx * l - 0.5, yy = gy * l - 0.5
    let feet: any Geometry3D = onlyCorners
        ? patternLinear(x: gx / dbnx, y: gy / dbny, spacingX: dbnx * l, spacingY: dbny * l,
                        blockBase(nx: dbnx, ny: dbny, length: l, styleHole: 0, offset: offset))
            .subtracting {
                patternLinear(x: 2, y: 2, spacingX: (gx - 1) * lGrid + dHole, spacingY: (gy - 1) * lGrid + dHole,
                              blockBaseHole(style: styleHole, offset: offset))
            }
        : patternLinear(x: gx / dbnx, y: gy / dbny, spacingX: dbnx * l, spacingY: dbny * l,
                        blockBase(nx: dbnx, ny: dbny, length: l, styleHole: styleHole, offset: offset))
    return Union {
        if finalCut {
            roundedRectangle(length: xx + 0.002, width: yy + 0.002, height: hBot / 1.5, radius: rFo1 / 2 + 0.001)
                .translated(z: hBase)
        }
        (finalCut
            ? feet.intersecting {
                roundedRectangle(length: xx + 0.005, width: yy + 0.005, height: hBase + hBot / 2 * 10, radius: rFo1 / 2 + 0.001)
                    .translated(z: -1)
            }
            : feet)
    }
}
```

Note the OpenSCAD fidelity points: the final-cut block is a separate union piece; `dbnx` multiplies `l` in the pattern spacing and in `blockBase`'s foot size (`nx: dbnx`); `block_base` uses the foot length `dbnx * l`. `blockBase` with `style_hole == 4` places `refinedHole()` translated by `(l/2 - dHoleFromSide, l/2 - dHoleFromSide, hSlit * 2)` and the other styles `blockBaseHole(style:offset:)` at z 0, all inside `patternCircular(count: 4)`.

```swift
struct BaseplateCutters {
    let d: Dimensions
    let b: BaseplateDimensions

    /// `cutter_weight`: a square pocket and four slot-and-round pockets.
    func weight() -> any Geometry3D {
        Union {
            Rectangle(b.bpCutSize).aligned(at: .center).extruded(height: b.bpCutDepth * 2).aligned(at: .centerZ)
            patternCircular(count: 4,
                Union {
                    Rectangle([b.bpRcutWidth, b.bpRcutLength]).aligned(at: .center)
                    Circle(diameter: b.bpRcutWidth).translated(y: b.bpRcutLength / 2)
                }
                .extruded(height: b.bpRcutDepth * 2).aligned(at: .centerZ)
                .translated(y: 10))
        }
    }

    /// `profile_skeleton`: the plate floor outline with bosses around the holes, in 2D.
    func skeletonProfile() -> any Geometry2D {
        let l = d.lGrid - 2 * d.rC2 - 2 * d.rC1
        let side = l - 2 * b.rSkel + 2 * d.dClear
        let boss = Rectangle(l)
            .offset(amount: d.rHole2 + b.rSkel + 2, style: .round)
            .translated(x: d.lGrid / 2 - d.dHoleFromSide, y: d.lGrid / 2 - d.dHoleFromSide)
        return Rectangle(side).aligned(at: .center)
            .subtracting { patternCircular2D(count: 4, boss) }
            .offset(amount: b.rSkel, style: .round)
    }
    // countersink, counterbore, holePattern, screwTogether: transcribe lines 150 to 209
}
```

`patternCircular2D` is a small private 2D helper (rotate copies in 2D), written in this file. The countersink is `Cylinder(radius: r_hole1 + d_clear, height: 100 * hBase)` centered plus a `convexHull()` of two cylinders (`Cylinder(radius: ..., height: dCs + 10)` and the second translated up by `dCs` with radius `+ dCs`) mirrored in z and translated by `dCs`. The counterbore is the through cylinder plus `Cylinder(radius: rCb, height: 2 * (hCb + 0.2))` centered, minus `copyMirror(Box([rCb * 3, rCb * 3, 10]).translated(x: -1.5 * rCb, y: rHole1 + dClear + 0.1, z: hCb - hSlit), along: .y)`. `holePattern` places four copies of its children at `(l/2 - dHoleFromSide, l/2 - dHoleFromSide)` through `patternCircular(count: 4, ...)`. `screwTogether` is `screw(gx, gy)` plus `screw(gy, gx)` rotated 90 degrees, where `screw(a, b)` is `copyMirror` (x) of a translate of `a * l / 2, 0, -off / 2` containing `patternLinear(x: 1, y: b, spacingX: 1, spacingY: l)` containing `patternLinear(x: 1, y: n, spacingX: 1, spacingY: dScrewHead + spacing)` containing a cylinder of diameter `dScrew`, height `l / 2`, rotated 90 degrees about y and centered.

```swift
public struct BaseplateDesign: Sendable {
    let c: BaseplateConfiguration
    let d = Dimensions.standard
    let b = BaseplateDimensions.standard
    public init(_ configuration: BaseplateConfiguration) { c = configuration }

    var screwTogether: Bool { c.plateStyle == .screwTogether || c.plateStyle == .screwTogetherMinimal }
    public var cellsX: Int { c.gridX == 0 ? Int((c.drawerX / d.lGrid).rounded(.down)) : c.gridX }
    public var cellsY: Int { c.gridY == 0 ? Int((c.drawerY / d.lGrid).rounded(.down)) : c.gridY }
    public var width: Double { max(Double(cellsX) * d.lGrid - b.bpXYClearance, c.drawerX) }
    public var depth: Double { max(Double(cellsY) * d.lGrid - b.bpXYClearance, c.drawerY) }

    /// `calculate_off`: how far the plate extends below z = 0.
    public var floorDepth: Double {
        if screwTogether { return 6.75 }
        switch c.plateStyle {
        case .thin: return 0
        case .weighted: return b.bpHBot
        default:
            return b.hSkel + (c.magnetHoles ? d.hHole : 0)
                + (c.screwHoles == .none ? c.screwDiameter : c.screwHoles == .countersink ? b.dCs : b.hCb)
        }
    }
    public var offsetX: Double { c.drawerX < width ? 0 : (Double(cellsX) * d.lGrid - b.bpXYClearance - c.drawerX) / 2 * c.fitX * -1 }
    public var offsetY: Double { c.drawerY < depth ? 0 : (Double(cellsY) * d.lGrid - b.bpXYClearance - c.drawerY) / 2 * c.fitY * -1 }
    // geometry(): see below
}
```
